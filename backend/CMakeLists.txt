cmake_minimum_required(VERSION 3.20)
project(mistakebook_native VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(PkgConfig REQUIRED)
pkg_check_modules(SQLITE REQUIRED IMPORTED_TARGET sqlite3>=3.34)
pkg_check_modules(SODIUM REQUIRED IMPORTED_TARGET libsodium)
find_package(nlohmann_json 3.10 REQUIRED)
find_package(Threads REQUIRED)
add_library(mistakebook_core src/core.cpp src/content.cpp src/notes.cpp src/print.cpp)
target_include_directories(mistakebook_core PUBLIC include)
target_link_libraries(mistakebook_core PUBLIC PkgConfig::SQLITE nlohmann_json::nlohmann_json Threads::Threads)
target_compile_options(mistakebook_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(mistakebook src/main.cpp src/oauth.cpp)
target_include_directories(mistakebook PRIVATE vendor)
target_link_libraries(mistakebook PRIVATE mistakebook_core PkgConfig::SODIUM Threads::Threads)
target_compile_options(mistakebook PRIVATE -Wall -Wextra -Wpedantic)
include(CTest)
if(BUILD_TESTING)
  add_executable(content_tests tests/content_tests.cpp)
  target_link_libraries(content_tests PRIVATE mistakebook_core)
  add_test(NAME content COMMAND content_tests)
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/notes_print_tests.cpp")
    add_executable(notes_print_tests tests/notes_print_tests.cpp)
    target_link_libraries(notes_print_tests PRIVATE mistakebook_core)
    add_test(NAME notes_print COMMAND notes_print_tests)
  endif()
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/embedding_benchmark.cpp")
    add_executable(embedding_benchmark tests/embedding_benchmark.cpp)
    target_link_libraries(embedding_benchmark PRIVATE mistakebook_core)
  endif()
endif()
install(TARGETS mistakebook RUNTIME DESTINATION bin)
